const nextVersion = (old: string, now: number) => new Date(Math.max(now, Date.parse(old) + 1)).toISOString();

const local = { writing: 'Device edit' };
const oldVersion = '2030-01-01T00:00:00.123456Z';
const newVersion = nextVersion(oldVersion, Date.parse('2026-10-02T00:00:00Z'));
async function reset(data = before, stamp = oldVersion) { await db.exec('truncate diagnostic_rows'); await db.query('insert into diagnostic_rows values ($1,$2::jsonb,$3::timestamptz)', ['A', JSON.stringify(data), stamp]); }
async function row() { return (await db.query('select data, updated_at::text as updated_at from diagnostic_rows where user_id=$1', ['A'])).rows[0] as any; }
