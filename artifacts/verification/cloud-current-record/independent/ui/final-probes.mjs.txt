function history(file, text) {
  const call = find(file, n => ts.isCallExpression(n) && n.expression.getText(asts[file]) === 't' && n.arguments[0]?.text === text);
  let ancestor = call;
  while (ancestor && !(ts.isJsxExpression(ancestor) && ts.isBinaryExpression(ancestor.expression)
    && ancestor.expression.operatorToken.kind === ts.SyntaxKind.AmpersandAmpersandToken)) ancestor = ancestor.parent;
  assert.ok(ancestor, 'History visibility expression missing');
  return { label: call.getText(asts[file]), visible: ancestor.expression.left.getText(asts[file]) };
}
