function find(file, predicate) {

  visit(asts[file]); assert.ok(result, `Missing AST target in ${file}`); return result;
}
function variable(file, name) {
  const node = find(file, n => ts.isVariableDeclaration(n) && n.name.getText(asts[file]) === name);
  locations[`${file}:${name}`] = asts[file].getLineAndCharacterOfPosition(node.getStart()).line + 1;
  return node.initializer;
}
