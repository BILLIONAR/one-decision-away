const clone = <T,>(value: T): T => JSON.parse(JSON.stringify(value));
const reads: string[] = [];
const writes: any[] = [];
const attempts: { kind: string }[] = [];
let readFailed = false;
let authCallback: ((event: string, session: any) => void) | undefined;
let app: any;
const crud = cloudCrudFixture({
  read: async id => {
    reads.push(id);
    return readFailed ? { data: null, error: { message: backupCopy(getLocale()).cloudUploadReadFailed } }
      : { data: clone(crud.rows.get(id) ?? null), error: null };
  },
  write: async row => { writes.push(clone(row)); return { error: null }; },
});
const from = crud.client.from;
const client = {
  ...crud.client,
  from: (...args: any[]) => {
    const query = (from as any)(...args);
    return { ...query,
      insert: (row: any) => { attempts.push({ kind: 'insert' }); return query.insert(row); },
      update: (row: any) => { attempts.push({ kind: 'update' }); return query.update(row); },
    };
  },
  auth: {
    getSession: async () => ({ data: { session: null }, error: null }),
    onAuthStateChange: (callback: typeof authCallback) => { authCallback = callback; return { data: { subscription: { unsubscribe() {} } } }; },
    signOut: async () => { authCallback?.('SIGNED_OUT', null); return { error: null }; },
    signInWithOtp: async () => { boundaryAttempts.authDelivery++; throw new Error('Auth delivery is outside this fixture.'); },
    verifyOtp: async () => { boundaryAttempts.authVerification++; throw new Error('Auth verification is outside this fixture.'); },
  },
};
function data(name: string) {
  const result = clone(getInitialDemoState());
  Object.assign(result.profile, { displayName: name, onboardingStep: 'completed', theme: 'light', soundMuted: true, nudgesEnabled: false, dailyWisdomEnabled: false });
  Object.assign(result, { [REPLACEMENT_EPOCH_KEY]: `synthetic-${name.replace(/\s+/g, '-').toLowerCase()}` });
  return result;
}

const remote = data('Synthetic existing cloud record');
localStorage.clear(); // Dedicated loopback fixture origin contains only synthetic data.

crud.rows.set('synthetic-account-A', { user_id: 'synthetic-account-A', data: remote, updated_at: '2030-01-01T00:00:00.000Z' });

  copy: () => backupCopy(getLocale()),
