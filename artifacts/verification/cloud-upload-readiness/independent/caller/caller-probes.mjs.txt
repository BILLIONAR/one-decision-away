let callback;
const handlers = {};
function visit(node) {
  if (ts.isVariableDeclaration(node) && node.name.getText(ast) === 'syncFromCloud') {
    assert.ok(ts.isCallExpression(node.initializer));
    callback = node.initializer.arguments[0].getText(ast);
  }
  if (ts.isJsxSelfClosingElement(node) && node.tagName.getText(ast) === 'BackupRestoreReview') {
    for (const attribute of node.attributes.properties) {
      if (ts.isJsxAttribute(attribute) && ['onCancel', 'onConfirm'].includes(attribute.name.getText(ast))) {
        handlers[attribute.name.getText(ast)] = attribute.initializer.expression.getText(ast);
      }
    }
  }
  ts.forEachChild(node, visit);
}
visit(ast);
assert.ok(callback && handlers.onCancel && handlers.onConfirm);

const deferred = () => {
  let resolve;
  let reject;
  const promise = new Promise((res, rej) => { resolve = res; reject = rej; });
  return { promise, resolve, reject };
};

function fixture(options = {}) {
  const events = [];
  const remote = { synthetic: 'reviewed cloud record' };
  const local = { synthetic: 'current device record' };
  const state = {
    accountCurrent: true, operationCurrent: true, durable: false,
    fallbackCalls: 0, pullCalls: 0, reviewCalls: 0, replaceCalls: 0,
    ackCalls: 0, refreshCalls: 0, toasts: [], events,
  };
  const heldRead = deferred();
  const context = {
    cloudReviewActive: { current: false }, cloudRestoreReview: null,
    data: local, Promise, fetch: denyFetch,
    getLocale: () => options.locale || 'en', backupCopy,
    t: value => value,
    setCloudRestoreReview(value) {
      context.cloudRestoreReview = value;
      if (value) { state.reviewCalls += 1; events.push('review opened'); }
      else events.push('review closed');
    },
    showToast(message, type) { state.toasts.push({ message, type }); events.push(`toast:${type}`); },
    async refreshData(value) {
      state.refreshCalls += 1;
      assert.equal(value.skipCloudPull, true);
      assert.deepEqual(Object.keys(value), ['skipCloudPull']);
      assert.equal(state.durable, true);
      events.push('refresh device without cloud pull');
      if (options.staleAfterRefresh) state.accountCurrent = false;
    },
    cloudSync: {
      currentAccountGuard: () => () => state.accountCurrent,
      currentOperationGuard() {
        events.push('capture operation');
        if (options.captureThrows) throw new Error('Synthetic storage read failure');
        return () => state.accountCurrent && state.operationCurrent;
      },
      async pullIfNewer(record, value) {
        state.pullCalls += 1;
        assert.equal(record, local);
        assert.equal(value.forReview, true);
        assert.deepEqual(Object.keys(value), ['forReview']);
        events.push('manual cloud read');
        if (options.readThrows) {
          throw new Error('Synthetic read failure');
        }
        if (options.holdRead) return heldRead.promise;
        if (options.staleRead) state.operationCurrent = false;
        return options.noRemote ? null : remote;
      },
      canApplyRemote(record, original) {
        assert.equal(record, remote);
        assert.equal(original, 'synthetic-original-bytes');
        events.push('replacement guard checked');
        return state.accountCurrent && state.operationCurrent && !options.replaceRejected;
      },
      async markRemoteApplied(record) {
        assert.equal(record, remote);
        assert.equal(state.durable, true);
        state.ackCalls += 1;
        events.push('ack after durable replacement');
        if (options.staleAfterAck) state.accountCurrent = false;
        return options.acknowledged !== false;
      },
      getState: () => ({ error: options.cloudError || null }),
    },
    repository: {
      async replaceAll(record, guard) {
        state.replaceCalls += 1;
        assert.equal(record, remote);
        assert.equal(typeof guard, 'function');
        events.push('replacement requested');
        if (options.replaceThrows) throw new Error('Synthetic durable commit failure');
        if (!guard('synthetic-original-bytes')) return false;
        state.durable = true;
        events.push('durable replacement');
        return true;
      },
    },
  };
  vm.createContext(context);
  vm.runInContext(executable, context);
  const runWithFallback = async () => {
    const handled = await context.run();
    if (!handled) state.fallbackCalls += 1;
    return handled;
  };
  return { state, context, remote, heldRead, runWithFallback };
}

async function waitForReview(f) {
  for (let count = 0; count < 25; count += 1) {
    if (f.context.cloudRestoreReview) return;
    await Promise.resolve();
  }
  throw new Error('Expected review did not open');
}
const report = {
  evidence: 'Actual extracted syncFromCloud and actual cloud modal confirm/cancel handlers executed in a VM with injected synthetic stubs. Actual locale copy imported. No mounted React, SDK, provider, database or native device.',
  beforeSourceSha256: before, cases: [], failures: [], fetchCalls: 0,
};
async function check(name, options, action) {
  const f = fixture(options);
  try {
    await action(f);
    assert.equal(f.context.cloudReviewActive.current, false, 'review single-flight ref must clear');
    report.cases.push({ name, passed: true, ...f.state });
  } catch (error) {
    report.failures.push({ name, error: String(error.stack || error) });
    report.cases.push({ name, passed: false, ...f.state });
  }
}

const noApply = f => {
  assert.equal(f.state.replaceCalls, 0); assert.equal(f.state.ackCalls, 0);
  assert.equal(f.state.refreshCalls, 0); assert.equal(f.state.durable, false);
};
