type Outcome = { ok?: boolean; message?: string; reject?: boolean; hold?: boolean };
let native = true;
let route = '/app/settings';
let revision = 0;
let epoch = 0;
let cloud = { configured: true, session: null as any, lastSyncAt: null, syncing: false, error: null, scopeRevision: 1 };
const cloudListeners = new Set<(state: typeof cloud) => void>();
const routeListeners = new Set<() => void>();
const calls: { kind: string; email?: string; code?: string; route?: string }[] = [];
const plans: Record<string, Outcome[]> = { send: [], verify: [] };
const pending: { kind: string; resolve: (outcome: Outcome) => void }[] = [];
const toasts: { text: string; kind: string }[] = [];
function routeEmit() { revision++; routeListeners.forEach(listener => listener()); }
function cloudEmit(patch: Record<string, unknown>) { cloud = { ...cloud, ...patch }; cloudListeners.forEach(listener => listener(cloud)); }
async function outcome(kind: string) {
  let plan = plans[kind].shift() ?? { ok: true, message: 'Synthetic response; no email was sent.' };
  if (plan.hold) plan = await new Promise<Outcome>(resolve => pending.push({ kind, resolve }));
  if (plan.reject) throw new Error('Synthetic rejected auth request.');
  return { ok: plan.ok ?? true, message: plan.message ?? 'Synthetic auth response.' };
}
