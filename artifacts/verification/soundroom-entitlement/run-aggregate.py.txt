from pathlib import Path
import hashlib,json,os,re,subprocess,time
root=Path('/workspace/oda-soundroom-entitlement-fix')
out=Path('/workspace/scratch/oda-soundroom-entitlement-fix')
out.mkdir(parents=True,exist_ok=True)
log=out/'aggregate-check-v2.log'
receipt=out/'build-receipt-v2.json'
assert not log.exists() and not receipt.exists()
base='8fe9e55b602cc757646aa2c51baac8e176284840'
old=json.loads((root/'artifacts/verification/cloud-upload-readiness/implementation/build-receipt-v3.json').read_text())
paths={row['file'] for row in old['source']}
for name in ['src','tests','scripts','public']:
    paths.update(str(p.relative_to(root)) for p in (root/name).rglob('*') if p.is_file())
paths=sorted(paths)
def snapshot():
    return [{'file':p,'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest()} for p in paths for b in [(root/p).read_bytes()]]
before=snapshot()
start=time.time()
env=dict(os.environ,VITE_BASE_PATH='/one-decision-away/')
with log.open('xb') as f:
    process=subprocess.run(['npm','run','check'],cwd=root,env=env,stdout=f,stderr=subprocess.STDOUT)
after=snapshot()
built=[{'file':str(p.relative_to(root)),'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest()} for p in sorted((root/'dist').rglob('*')) if p.is_file() for b in [p.read_bytes()]]
raw=log.read_bytes()
text=raw.decode(errors='replace')
passed=re.findall(r'^(?:# |ℹ )pass (\d+)\s*$',text,re.M)
failed=re.findall(r'^(?:# |ℹ )fail (\d+)\s*$',text,re.M)
changed=set(subprocess.check_output(['git','diff','--name-only',base],cwd=root,text=True).splitlines())|set(subprocess.check_output(['git','ls-files','--others','--exclude-standard'],cwd=root,text=True).splitlines())
r={'base':base,'command':'VITE_BASE_PATH=/one-decision-away/ npm run check','exitCode':process.returncode,'passed':int(passed[-1]) if passed else None,'failed':int(failed[-1]) if failed else None,'log':log.name,'logBytes':len(raw),'logSha256':hashlib.sha256(raw).hexdigest(),'elapsedSeconds':round(time.time()-start,3),'unchangedDuringRun':before==after,'changedSource':{row['file']:{k:v for k,v in row.items() if k!='file'} for row in before if row['file'] in changed},'source':before,'build':built}
receipt.write_text(json.dumps(r,indent=2)+'\n')
print(json.dumps({k:v for k,v in r.items() if k not in ['changedSource','source','build']},indent=2))
assert before==after,'Source changed during aggregate run'
assert process.returncode==0,'Aggregate failed; original log preserved'
