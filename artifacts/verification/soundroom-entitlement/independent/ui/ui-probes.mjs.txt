function fixture({ sectionId = 'relax', index = 0, current = access({}), locked = false, playing = false,
  retained = null, timer = 30 } = {}) {
  const section = SOUND_ROOM_SECTIONS.find(item => item.id === sectionId);
  const sound = section.sounds[index];
  const result = { routes: [], toggleCalls: [], pauseCalls: 0, resumeCalls: 0,
    simulatedStarts: 0, boundaryDenials: 0, timers: [] };
  const live = { access: current, retained, timer };
  const context = {
    sound, section, sectionId, indexInCategory: index, i: index, locked,
    pro: { ...current, gating: current.available },
    player: { playing, track: retained?.track || null, timerMinutes: timer },
    isSoundLocked,
    purchases: { getState: () => live.access },
    setActiveRoute: route => result.routes.push(route),
    soundRoomPlayer: {
      toggle(track, options) {
        result.toggleCalls.push({ track, options: { ...options } });
        const selection = resolveSoundRoomAccess(track, options);
        if (!selection || !canPlaySoundRoom(selection.restricted, live.access)) {
          result.boundaryDenials += 1; return false;
        }
        result.simulatedStarts += 1; return true;
      },
      pause() { result.pauseCalls += 1; },
      resume(...args) {
        assert.equal(args.length, 0, 'bar delegates to retained player selection');
        result.resumeCalls += 1;
        if (!live.retained) { result.boundaryDenials += 1; return false; }
        const selected = resolveSoundRoomAccess(live.retained.track, live.retained.origin || {});
        if (!selected || !canPlaySoundRoom(selected.restricted, live.access)) {
          result.boundaryDenials += 1; return false;
        }
        result.simulatedStarts += 1; return true;
      },
      setTimer(minutes) { result.timers.push(minutes); live.timer = minutes; context.player.timerMinutes = minutes; },
    },
  };
  vm.createContext(context); vm.runInContext(executable, context);
  return { context, result, live, sound };
}
